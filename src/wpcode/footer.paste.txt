/* --- 0. Analytics: conversion events + first-touch attribution --- */
/* GA4 is Site Kit's tag (window.gtag). Every event carries page_path and, for clicks,
 * cta_location = the id of the nearest enclosing section/element with an id, else its class. */
window.pacerTrack = window.pacerTrack || function(name, params) {
  try {
    params = params || {};
    if (!params.page_path) params.page_path = window.location.pathname;
    if (typeof window.gtag === 'function') window.gtag('event', name, params);
  } catch (e) {}
};

/* GA4 user_id: after a demo signup the Worker hands back an opaque visitor id (pv_ + 16 hex, never an
 * email). It is kept in localStorage and set on every later page, so GA4 links that person's visits across
 * sessions. The D1 lead row (request_id ending in the same 16 hex) maps it to the Apollo contact. */
window.pacerSetUser = window.pacerSetUser || function(uid) {
  if (!/^pv_[0-9a-f]{16}$/.test(uid || '')) return;
  try { window.localStorage.setItem('pacer_uid', uid); } catch (e) {}
  try { if (typeof window.gtag === 'function') window.gtag('set', { user_id: uid }); } catch (e) {}
};
(function() { var u = null; try { u = window.localStorage.getItem('pacer_uid'); } catch (e) {} if (u) window.pacerSetUser(u); })();

/* First touch: the visitor's first UTMs / referrer / landing page, kept 30 days in
 * localStorage. A capture uses the current URL's UTMs when present, else the first touch;
 * an outside referrer with no UTMs becomes utm_source=<host>, utm_medium=referral. */
window.pacerAttr = window.pacerAttr || (function() {
  var KEY = 'pacer_ft', TTL = 30 * 864e5, now = Date.now();
  var qp = new URLSearchParams(window.location.search), ft = null;
  try { ft = JSON.parse(window.localStorage.getItem(KEY) || 'null'); } catch (e) {}
  if (!ft || now - (ft.ts || 0) > TTL) {
    var ref = '';
    try { ref = document.referrer ? new URL(document.referrer).hostname : ''; } catch (e) {}
    if (ref === window.location.hostname) ref = '';
    ft = { utm_source: qp.get('utm_source') || (ref || ''), utm_medium: qp.get('utm_medium') || (ref ? 'referral' : ''),
           utm_campaign: qp.get('utm_campaign') || '', landing: window.location.pathname, ts: now };
    try { window.localStorage.setItem(KEY, JSON.stringify(ft)); } catch (e) {}
  }
  return function() {
    var cur = qp.get('utm_source');
    return {
      utm_source: cur || ft.utm_source || null,
      utm_medium: cur ? (qp.get('utm_medium') || null) : (ft.utm_medium || null),
      utm_campaign: cur ? (qp.get('utm_campaign') || null) : (ft.utm_campaign || null),
      landing: ft.landing || null
    };
  };
})();

(function() {
  if (window.__pacerClicks) return; window.__pacerClicks = 1;
  function where(el) {
    for (var n = el; n && n !== document.body; n = n.parentElement) {
      if (n.id && n.id.indexOf('pacerai-') !== 0) return n.id;
      if (n.tagName === 'NAV' || n.tagName === 'FOOTER') return n.tagName.toLowerCase();
      if (n.tagName === 'SECTION' && n.className) return String(n.className).split(' ')[0];
    }
    return el.className ? String(el.className).split(' ')[0] : '';
  }
  document.addEventListener('click', function(e) {
    var a = e.target && e.target.closest ? e.target.closest('a[href]') : null;
    if (!a) return;
    var href = a.getAttribute('href') || '', name = '', extra = {};
    var url = null;
    try { url = new URL(href, window.location.href); } catch (err) {}
    var samePage = url && url.hash.length > 1 && url.origin === window.location.origin &&
      url.pathname.replace(/\/$/, '') === window.location.pathname.replace(/\/$/, '');
    /* The claude.ai install link (/demo-connect hero and step 1): a step toward the connector sign-in, not a key
     * event. cta_location comes from the link's data-cta ("hero" / "steps"). */
    if (href.indexOf('claude.ai/customize/connectors') !== -1) { name = 'install_click'; extra.cta_location = a.getAttribute('data-cta') || 'unknown'; }
    else if (href.indexOf('calendly.com') !== -1) name = 'schedule_walkthrough';
    else if (href.indexOf('/demo-connect') !== -1 && window.location.pathname.indexOf('/demo-connect') !== 0) name = 'try_demo_click';
    else if (href.indexOf('/tear-sheet') !== -1) name = 'tear_sheet_open';
    else if (a.getAttribute('data-filter')) { name = 'resource_filter'; extra.category = a.getAttribute('data-filter'); }
    else if (samePage) { name = 'section_jump'; extra.section = url.hash.slice(1); }
    if (!name) return;
    var params = { cta_location: where(a), link_text: (a.textContent || '').trim().slice(0, 60), link_url: href };
    for (var k in extra) params[k] = extra[k];
    window.pacerTrack(name, params);
  }, true);

  /* Arriving on a #hash (a shared /resources/#arr-snowballs link, or /#pricing from another page). */
  if (window.location.hash.length > 1) {
    var h = decodeURIComponent(window.location.hash.slice(1)).slice(0, 60);
    if (window.location.pathname.indexOf('/resources') === 0 && document.querySelector('[data-filter="' + h.replace(/"/g, '') + '"]'))
      window.pacerTrack('resource_filter', { category: h, cta_location: 'arrival' });
    else window.pacerTrack('section_jump', { section: h, cta_location: 'arrival' });
  }

  /* Worker-served iframes (signup card, demo videos, pricing estimator) report events with
   * postMessage({type:'pacer:event', name, params}). Only that origin is trusted. */
  var WORKERS = { 'https://pacer-demo-worker.will-078.workers.dev': 1,           /* signup card, homepage video */
                  'https://pacer-demo-worker-staging.will-078.workers.dev': 1 };  /* the /demo-connect/ reel */

                  demo_video_complete: 1, demo_video_replay: 1, demo_video_review: 1 };
  window.addEventListener('message', function(e) {
    if (!WORKERS[e.origin] || !e.data || e.data.type !== 'pacer:event' || !ALLOWED[e.data.name]) return;
    var p = {}, src = e.data.params || {};
    if (e.data.name === 'demo_signup' && src.user_id) window.pacerSetUser(String(src.user_id));
    ['cta_location', 'video', 'plan', 'value'].forEach(function(k) { if (src[k] != null) p[k] = String(src[k]).slice(0, 80); });
    window.pacerTrack(e.data.name, p);
  });
})();

/* --- 1. Homepage hero typed-line animation --- */
(function(){
  var el = document.querySelector('#pacerai-homepage .typed-line');
  var cur = document.querySelector('#pacerai-homepage .type-cursor');
  if (!el || !cur) return;
  var ph = ['Board Reporting.', 'Operational Cadence.', 'Sales Strategy.',
    'Due Diligence.'];
  var pi = 0;
  var ci = 0;
  var del = false;
  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    el.textContent = ph[0];
    cur.style.display = 'none';
    return;
  }
  function tick() {
    var w = ph[pi];
    if (!del) {
      el.textContent = w.substring(0, ci + 1);
      ci++;
      if (ci === w.length) {
        del = true;
        setTimeout(tick, 2000);
        return;
      }
      setTimeout(tick, 60);
    } else {
      el.textContent = w.substring(0, ci - 1);
      ci--;
      if (ci === 0) {
        del = false;
        pi = (pi + 1) % ph.length;
        setTimeout(tick, 400);
        return;
      }
      setTimeout(tick, 35);
    }
  }
  setTimeout(tick, 800);
})();

(function() {
  /* 2. Lead-capture form handler (all .wp-cta-form on the page) */
  /* Each form declares its own offer:
   *   data-asset-slug   which ASSETS entry the Worker should serve (default: the white paper)
   *   data-success-msg  confirmation copy for this placement
   *   data-redirect     where to send the visitor after a capture-only submit
   * A response with download_url opens the file; one without just confirms. */
  var WORKER_URL = 'https://whitepaper-worker.will-078.workers.dev';

  document.querySelectorAll('.wp-cta-form').forEach(function(form) {
    var input = form.querySelector('input[type="email"]');
    var button = form.querySelector('button');
    if (!input || !button) return;

    var origText = button.textContent;
    var slug = form.getAttribute('data-asset-slug') || 'board-quality-arr-snowballs';
    var successMsg = form.getAttribute('data-success-msg') || 'Check your inbox \u2014 we\u2019ll send you a copy too.';
    var redirect = form.getAttribute('data-redirect') || '';

    var msgEl = document.createElement('p');
    msgEl.style.cssText = 'font-size:12px;margin-top:8px;text-align:center;min-height:18px;';
    form.parentNode.insertBefore(msgEl, form.nextSibling);

    form.addEventListener('submit', function(e) {
      e.preventDefault();
      msgEl.textContent = '';
      msgEl.style.color = '#B03A3A';

      var email = (input.value || '').trim();
      if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) {
        msgEl.textContent = 'Please enter a valid work email.';
        return;
      }

      button.disabled = true;
      button.textContent = 'One moment\u2026';

      var at = window.pacerAttr();
      var body = JSON.stringify({
        email: email,
        asset_slug: slug,
        page_url: window.location.origin + window.location.pathname,
        utm_source: at.utm_source,
        utm_medium: at.utm_medium,
        utm_campaign: at.utm_campaign
      });

      fetch(WORKER_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: body
      })
      .then(function(r) { return r.json(); })
      .then(function(data) {
        if (!data || !data.success) {
          msgEl.textContent = 'Something went wrong. Please try again.';
          button.disabled = false;
          button.textContent = origText;
          return;
        }
        msgEl.style.color = '#15803D';
        msgEl.textContent = successMsg;
        window.pacerTrack(redirect ? 'demo_email_capture' : 'whitepaper_download', { asset_slug: slug });
        if (data.download_url) {
          button.textContent = 'Downloading\u2026';
          window.open(data.download_url, '_blank');
        } else {
          button.textContent = 'Thanks \u2014 you\u2019re in';
          if (redirect) {
            /* Carry the email forward so the demo widget doesn't ask twice — in sessionStorage,
             * never the URL (page URLs go to GA4, Apollo and Jetpack). */
            try { window.sessionStorage.setItem('pacer_email', email); } catch (e) {}
            setTimeout(function() { window.location.href = redirect; }, 900);
          }
        }
      })
      .catch(function() {
        msgEl.textContent = 'Network error. Please try again.';
        button.disabled = false;
        button.textContent = origText;
      });
    });
  });

  /* 2b. Demo signup iframe — pass the email (from the hero form, via sessionStorage; legacy
   * ?email= links still work) and the visitor's attribution into the widget. The widget is
   * cross-origin (Worker-served), so the parent cannot fill it directly; the Worker reads
   * the params and prefills / records them server-side instead. */
  (function() {
    var frames = document.querySelectorAll('iframe[src*="/signup"]');
    if (!frames.length) return;
    var em = '';
    try { em = window.sessionStorage.getItem('pacer_email') || ''; } catch (e) {}
    if (!em) em = (new URLSearchParams(window.location.search).get('email') || '').trim();
    var at = window.pacerAttr(), q = [];
    if (em) q.push('email=' + encodeURIComponent(em));
    ['utm_source', 'utm_medium', 'utm_campaign', 'landing'].forEach(function(k) { if (at[k]) q.push(k + '=' + encodeURIComponent(at[k])); });
    q.push('page=' + encodeURIComponent(window.location.pathname));
    frames.forEach(function(f) {
      if (f.src.indexOf('page=') !== -1) return;
      f.src = f.src + (f.src.indexOf('?') === -1 ? '?' : '&') + q.join('&');
    });
  })();

  /* 3. Pipeline number stream animation */
  var c = document.getElementById('num-stream-lt');
  if (c && !window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    var nums = ['$24,035','$81,115','$88,980','$118,795','106.7%','NRR 101.3%','GRR 91.3%','$197,545','$510,470','29.4%'];
    function spawn() {
      var el = document.createElement('span');
      el.className = 'num-particle-lt';
      el.textContent = nums[Math.floor(Math.random() * nums.length)];
      el.style.top = (Math.random() * 280) + 'px';
      el.style.animationDuration = (8 + Math.random() * 6) + 's';
      el.style.fontSize = (9 + Math.random() * 4) + 'px';
      c.appendChild(el);
      el.addEventListener('animationend', function() { el.remove(); });
    }
    for (var i = 0; i < 6; i++) setTimeout(spawn, i * 1500);
    setInterval(spawn, 2000);
  }

  /* 4. Mobile nav accordion */
  if (window.innerWidth <= 768) {
    document.querySelectorAll('#pacerai-homepage .nav-links > li > a').forEach(function(a) {
      var li = a.parentElement;
      if (!li.querySelector('.dropdown')) return;
      a.addEventListener('click', function(e) {
        e.preventDefault();
        li.classList.toggle('mobile-expanded');
      });
    });
  }
})();

/* --- 5. Research waitlist form handler (/research) --- */
/* POSTs to the same Cloudflare Worker as the white-paper form, but with
 * asset_slug 'research-waitlist'. No download is returned — on success it
 * just confirms the signup. Worker must accept this asset_slug and respond
 * { success: true } (Apollo tag + Slack ping). Safe site-wide: exits if the
 * form isn't on the page. */
(function() {
  var form = document.querySelector('.research-waitlist-form');
  if (!form) return;

  var WORKER_URL = 'https://whitepaper-worker.will-078.workers.dev';

  var input = form.querySelector('input[type="email"]');
  var button = form.querySelector('button');
  var origText = button.textContent;

  var msgEl = form.parentNode.querySelector('.waitlist-msg');
  if (!msgEl) {
    msgEl = document.createElement('p');
    msgEl.className = 'waitlist-msg';
    form.parentNode.insertBefore(msgEl, form.nextSibling);
  }

  form.addEventListener('submit', function(e) {
    e.preventDefault();
    msgEl.textContent = '';
    msgEl.style.color = '#C94C4C';

    var email = (input.value || '').trim();
    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) {
      msgEl.textContent = 'Please enter a valid work email.';
      return;
    }

    button.disabled = true;
    button.textContent = 'Joining…';

    var at = window.pacerAttr();
    var body = JSON.stringify({
      email: email,
      asset_slug: 'research-waitlist',
      page_url: window.location.origin + window.location.pathname,
      utm_source: at.utm_source,
      utm_medium: at.utm_medium,
      utm_campaign: at.utm_campaign
    });

    fetch(WORKER_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: body
    })
    .then(function(r) { return r.json(); })
    .then(function(data) {
      if (data && data.success) {
        window.pacerTrack('research_waitlist', {});
        button.textContent = 'You’re on the list';
        msgEl.style.color = '#2DB87A';
        msgEl.textContent = 'You’re on the list — we’ll email you the day we launch.';
        input.disabled = true;
      } else {
        msgEl.textContent = 'Something went wrong. Please try again.';
        button.disabled = false;
        button.textContent = origText;
      }
    })
    .catch(function() {
      msgEl.textContent = 'Network error. Please try again.';
      button.disabled = false;
      button.textContent = origText;
    });
  });
})();

(function() {
  var rotor = document.querySelector('#pacerai-homepage .pa-rotor');
  if (!rotor) return;
  if (window.__paRotor) return; window.__paRotor = 1; /* guard: page injector may own this */

  /* The page owns the phrase list (data-phrases on .pa-rotor, "|"-separated) so a copy change never
     needs a WPCode paste. The fallback below only runs if that attribute is missing. */
  var phrases = (rotor.getAttribute('data-phrases') || '').split('|').filter(Boolean);
  if (phrases.length < 2) phrases = ['make plan with confidence.', 'close the gap to plan.',
    'improve Net Retention Rates.', 'cross-sell products.', 'improve durable revenue growth.',
    'improve forecast accuracy.'];
  var i = 0;
  setInterval(function() {
    i = (i + 1) % phrases.length;
    rotor.style.opacity = '0';
    setTimeout(function() {
      rotor.textContent = phrases[i];
      rotor.style.transition = 'opacity .35s ease';
      rotor.style.opacity = '1';
    }, 200);
  }, 2400);
})();
